mutex = sem(1)
cocina = sem(0)
M = 20;
//no respeta un orden cuando no hay mas comida.
enfermo(){
	while(true){
		mutex.wait();
		if(M == 0){
			cocina.signal();
		}else{
			tomar_porcion();
			M.getanddec();
			mutex.signal();	
			comer();
		}
	}
}

cocinero(){
	cocina.wait();
	M.getandAdd(20);
	mutex.signal();
}
